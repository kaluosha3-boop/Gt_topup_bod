import time

from aiogram.types import InlineKeyboardMarkup, InlineKeyboardButton, BotCommand
from aiogram.dispatcher.middlewares import BaseMiddleware
from aiogram.dispatcher.handler import CancelHandler

# استدعاء البيانات الحساسة من المتغيرات البيئية بأمان
API_TOKEN = os.getenv('BOT_TOKEN')
ADMIN_ID = os.getenv('ADMIN_ID')

    try:

    except ValueError:

bot = Bot(token=API_TOKEN)

# ==========================================
# 🛡️ 1. ANTI-SPAM & RATE LIMITING MIDDLEWARE
# ==========================================
class ThrottlingMiddleware(BaseMiddleware):
    def __init__(self, limit=1.5): # الحد المسموح: رسالة كل 1.5 ثانية
        self.rate_limit = limit
        self.user_timeouts = {}
        super(ThrottlingMiddleware, self).__init__()

    async def on_process_message(self, message: types.Message, data: dict):
        user_id = message.from_user.id
        current_time = time.time()

        if user_id in self.user_timeouts:
            last_time = self.user_timeouts[user_id]
            if current_time - last_time < self.rate_limit:
                await message.reply("⚠️ Too many requests! Please wait a moment.")
                raise CancelHandler()

        self.user_timeouts[user_id] = current_time

dp.middleware.setup(ThrottlingMiddleware())

# DATABASE INITIALIZATION
# ==========================================
def init_db():
    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS products (
            id TEXT PRIMARY KEY,
            price REAL,
            currency TEXT,
            name TEXT,
            category TEXT
        )
    ''')
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS settings (
            key TEXT PRIMARY KEY,
            value TEXT
        )
    ''')
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS orders (
            order_id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            username TEXT,
            product_id TEXT,
            product_name TEXT,
            price REAL,
            currency TEXT,
            status TEXT,
            player_id TEXT DEFAULT NULL
        )
    ''')
    conn.commit()
    conn.close()

init_db()

def is_admin(user_id: int) -> bool:
    return ADMIN_ID is not None and user_id == ADMIN_ID

    await bot.set_my_commands(commands)

@dp.callback_query_handler(lambda c: c.data.startswith('cat_'))
async def process_category_codes(callback_query: types.CallbackQuery):
    cat = callback_query.data.split('_')[1]
    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute("SELECT id, price, currency, name FROM products WHERE id LIKE ?", (f"{cat}-%",))
    products = cursor.fetchall()
    conn.close()

    keyboard = InlineKeyboardMarkup(row_width=1)
    if not products:
        keyboard.add(InlineKeyboardButton("⬅️ Back", callback_data="main_codes"))
        await callback_query.message.edit_text("No products available in this category currently.", reply_markup=keyboard)
        return

    for prod in products:
        prod_id, price, currency, name = prod
        keyboard.add(InlineKeyboardButton(f"{name} — {price} {currency}", callback_data=f"buy_{prod_id}"))
    
    keyboard.add(InlineKeyboardButton("⬅️ Back", callback_data="main_codes"))
    cat_name = "Free Fire" if cat == "ff" else "PUBG Mobile"
    await callback_query.message.edit_text(f"Select a product ({cat_name} Codes):", reply_markup=keyboard)

@dp.callback_query_handler(lambda c: c.data.startswith('buy_'))
async def process_buy(callback_query: types.CallbackQuery):
    prod_id = callback_query.data.split('_', 1)[1]
    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute("SELECT id, price, currency, name FROM products WHERE id = ?", (prod_id,))
    prod = cursor.fetchone()
    cursor.execute("SELECT value FROM settings WHERE key = 'payment'")
    payment = cursor.fetchone()
    
    if not prod:
        await callback_query.answer("Product not found.", show_alert=True)
        conn.close()
        return

    p_id, price, currency, name = prod
    user = callback_query.from_user
    cursor.execute("INSERT INTO orders (user_id, username, product_id, product_name, price, currency, status) VALUES (?, ?, ?, ?, ?, ?, 'pending')",
                   (user.id, user.username or user.first_name, p_id, name, price, currency))
    order_id = cursor.lastrowid
    conn.commit()
    conn.close()

    pay_text = payment[0] if payment else "Please contact admin (@GUODA36) to pay."
    msg = (
        f"Order #{order_id}: {name} — {price} {currency}\n\n"
        f"Payment Instructions:\n{pay_text}\n\n"
        "After transferring, send a photo of the receipt here for review."
    )
    await callback_query.message.edit_text(msg)

@dp.message_handler(content_types=types.ContentType.PHOTO)
async def handle_receipt(message: types.Message):
    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute("SELECT order_id, product_name FROM orders WHERE user_id = ? AND status = 'pending' ORDER BY order_id DESC LIMIT 1", (message.from_user.id,))
    order = cursor.fetchone()
    conn.close()

    if not order:
        await message.reply("No pending orders found. Please use /store first.")
        return

    order_id, prod_name = order
    await message.reply("Receipt sent to management for review. It will be processed soon.")

    if ADMIN_ID:
        caption = f"New Order for Review — #{order_id}\nUser: @{message.from_user.username or message.from_user.first_name} (ID: {message.from_user.id})\nProduct: {prod_name}"
        keyboard = InlineKeyboardMarkup()
        keyboard.add(
            InlineKeyboardButton("Approve & Deliver", callback_data=f"approve_{order_id}"),
            InlineKeyboardButton("Reject Order", callback_data=f"reject_{order_id}")
        )
        await bot.send_photo(ADMIN_ID, message.photo[-1].file_id, caption=caption, reply_markup=keyboard)

@dp.message_handler(commands=['addproduct'])
async def add_product(message: types.Message):
    if not is_admin(message.from_user.id):
        return
    parts = message.text.split(maxsplit=4)
    if len(parts) < 5:
        await message.reply("Usage: /addproduct <id> <price> <currency> <name>")
        return
    
    prod_id, price_str, currency, name = parts[1], parts[2], parts[3], parts[4]
    try:
        price = float(price_str)
    except ValueError:
        await message.reply("Price must be a valid number.")
        return

    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute("INSERT OR REPLACE INTO products (id, price, currency, name) VALUES (?, ?, ?, ?)",
                   (prod_id, price, currency, name))
    conn.commit()
    conn.close()
    await message.reply(f"Product saved successfully: {name}")

@dp.message_handler(commands=['deliver'])
async def deliver_code(message: types.Message):
    if not is_admin(message.from_user.id):
        return
    parts = message.text.split(maxsplit=2)
    if len(parts) < 3:
        await message.reply("Usage: /deliver <order_id> <details/code>")
        return
    order_id_str, details = parts[1], parts[2]
    try:
        order_id = int(order_id_str)
    except ValueError:
        await message.reply("Invalid order ID.")
        return

    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute("SELECT user_id, status FROM orders WHERE order_id = ?", (order_id,))
    res = cursor.fetchone()

    if not res or res[1] != 'approved':
        await message.reply("Order not found or not approved yet.")
        conn.close()
        return

    user_id = res[0]
    cursor.execute("UPDATE orders SET status = 'delivered' WHERE order_id = ?", (order_id,))
    conn.commit()
    conn.close()

    try:
        await bot.send_message(user_id, f"Your order has been completed:\n`{details}`", parse_mode="Markdown")
        await message.reply(f"Order #{order_id} fulfilled successfully.")
    except Exception as e:
        await message.reply(f"Failed to send to customer: {e}")

if __name__ == '__main__':
