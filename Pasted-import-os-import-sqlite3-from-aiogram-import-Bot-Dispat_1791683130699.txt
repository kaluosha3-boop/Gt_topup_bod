import os
import sqlite3
from aiogram import Bot, Dispatcher, executor, types
from aiogram.types import InlineKeyboardMarkup, InlineKeyboardButton

# استخراج التوكن ومعرف الأدمن من البيئة
API_TOKEN = os.getenv('BOT_TOKEN')
ADMIN_ID = os.getenv('ADMIN_ID')

if ADMIN_ID:
    ADMIN_ID = int(ADMIN_ID)

bot = Bot(token=API_TOKEN)
dp = Dispatcher(bot)

# إنشاء وتحديث قاعدة البيانات
def init_db():
    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS products (
            id TEXT PRIMARY KEY,
            price REAL,
            currency TEXT,
            name TEXT,
            category TEXT
        )
    ''')
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS settings (
            key TEXT PRIMARY KEY,
            value TEXT
        )
    ''')
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS orders (
            order_id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            username TEXT,
            product_id TEXT,
            product_name TEXT,
            price REAL,
            currency TEXT,
            status TEXT
        )
    ''')
    conn.commit()
    conn.close()

init_db()

def is_admin(user_id: int) -> bool:
    return user_id == ADMIN_ID

# --- أوامر البوت ---

@dp.message_handler(commands=['start', 'help'])
async def send_welcome(message: types.Message):
    msg = (
        "أهلاً بك في متجر أكواد الألعاب! 🎮\n\n"
        "استخدم /products لعرض أقسام المنتجات المتاحة.\n"
    )
    if is_admin(message.from_user.id):
        msg += (
            "\nإعداد الإدارة:\n"
            "/addproduct <id> <price> <currency> <name>\n"
            "(ملاحظة: استخدم id يبدأ بـ ff- لفري فاير وبـ pubg- لببجي)\n"
            "/setpayment <instructions>\n"
            "/orders لعرض الطلبات قيد المراجعة\n"
            "/deliver <order_id> <real_code>\n"
        )
    await message.reply(msg)

    keyboard.add(
        InlineKeyboardButton("🔥 Free Fire", callback_data="cat_ff"),

# معالجة اختيار القسم والعودة
@dp.callback_query_handler(lambda c: c.data.startswith('cat_') or c.data == 'back_to_cats')
async def process_category(callback_query: types.CallbackQuery):
    if callback_query.data == 'back_to_cats':
        keyboard = InlineKeyboardMarkup(row_width=2)
        keyboard.add(
            InlineKeyboardButton("🔥 Free Fire", callback_data="cat_ff"),
            InlineKeyboardButton("⚡ PUBG Mobile", callback_data="cat_pubg")
        )
        await callback_query.message.edit_text("اختر القسم الذي تريد الشراء منه:", reply_markup=keyboard)
        return

    cat = callback_query.data.split('_')[1] # ff or pubg
    
    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    # جلب المنتجات التي تبدأ بـ معرف القسم
    cursor.execute("SELECT id, price, currency, name FROM products WHERE id LIKE ?", (f"{cat}-%",))
    products = cursor.fetchall()
    conn.close()

    if not products:
        keyboard = InlineKeyboardMarkup()
        keyboard.add(InlineKeyboardButton("⬅️ العودة للأقسام", callback_data="back_to_cats"))
        await callback_query.message.edit_text("لا توجد منتجات متاحة في هذا القسم حالياً.", reply_markup=keyboard)
        return

    keyboard = InlineKeyboardMarkup(row_width=1)
    for prod in products:
        prod_id, price, currency, name = prod
        keyboard.add(InlineKeyboardButton(f"{name} — {price} {currency}", callback_data=f"buy_{prod_id}"))
    
    keyboard.add(InlineKeyboardButton("⬅️ العودة للأقسام", callback_data="back_to_cats"))
    
    cat_name = "Free Fire 🎮" if cat == "ff" else "PUBG Mobile 🔫"
    await callback_query.message.edit_text(f"اختر المنتج من قسم {cat_name}:", reply_markup=keyboard)

@dp.message_handler(commands=['addproduct'])
async def add_product(message: types.Message):
    if not is_admin(message.from_user.id):
        return
    parts = message.text.split(maxsplit=4)
    if len(parts) < 5:
        await message.reply("الاستخدام: /addproduct <id> <price> <currency> <name>\nمثال لفري فاير: /addproduct ff-110 45 EGP Free Fire 110 Diamonds\nمثال لببجي: /addproduct pubg-60 45 EGP PUBG Mobile 60 UC")
        return
    
    prod_id, price_str, currency, name = parts[1], parts[2], parts[3], parts[4]
    try:
        price = float(price_str)
    except ValueError:
        await message.reply("السعر يجب أن يكون رقماً.")
        return

    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute("INSERT OR REPLACE INTO products (id, price, currency, name) VALUES (?, ?, ?, ?)",
                   (prod_id, price, currency, name))
    conn.commit()
    conn.close()
    await message.reply(f"تم حفظ المنتج: {name}")

@dp.message_handler(commands=['setpayment'])
async def set_payment(message: types.Message):
    if not is_admin(message.from_user.id):
        return
    instructions = message.text.partition(' ')[2].strip()
    if not instructions:
        await message.reply("الاستخدام: /setpayment <تعليمات الدفع>")
        return
    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute("INSERT OR REPLACE INTO settings (key, value) VALUES ('payment', ?)", (instructions,))
    conn.commit()
    conn.close()
    await message.reply("تم حفظ تعليمات الدفع.")

# شراء المنتج
@dp.callback_query_handler(lambda c: c.data.startswith('buy_'))
async def process_buy(callback_query: types.CallbackQuery):
    prod_id = callback_query.data.split('_', 1)[1]
    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute("SELECT id, price, currency, name FROM products WHERE id = ?", (prod_id,))
    prod = cursor.fetchone()
    cursor.execute("SELECT value FROM settings WHERE key = 'payment'")
    payment = cursor.fetchone()
    
    if not prod:
        await callback_query.answer("المنتج غير موجود.", show_alert=True)
        conn.close()
        return

    p_id, price, currency, name = prod
    user = callback_query.from_user
    cursor.execute("INSERT INTO orders (user_id, username, product_id, product_name, price, currency, status) VALUES (?, ?, ?, ?, ?, ?, 'pending')",
                   (user.id, user.username or user.first_name, p_id, name, price, currency))
    order_id = cursor.lastrowid
    conn.commit()
    conn.close()

    pay_text = payment[0] if payment else "يرجى التواصل مع الأدمن للدفع."
    msg = (
        f"طلبك رقم {order_id}: {name} — {price} {currency}\n\n"
        f"تعليمات الدفع:\n{pay_text}\n\n"
        "بعد التحويل، أرسل صورة الإيصال هنا للمراجعة. لن يُرسل أي كود قبل مراجعة الطلب."
    )
    await callback_query.message.edit_text(msg)

# إرسال صورة الإيصال
@dp.message_handler(content_types=types.ContentType.PHOTO)
async def handle_receipt(message: types.Message):
    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute("SELECT order_id, product_name FROM orders WHERE user_id = ? AND status = 'pending' ORDER BY order_id DESC LIMIT 1", (message.from_user.id,))
    order = cursor.fetchone()
    conn.close()

    if not order:
        await message.reply("لا يوجد طلب معلق حالياً الشراء من /products أولاً.")
        return

    order_id, prod_name = order
    await message.reply("تم إرسال الإيصال إلى الإدارة للمراجعة. لن يُرسل الكود إلا بعد التحقق.")

    if ADMIN_ID:
        caption = f"طلب جديد للمراجعة — رقم {order_id}\nالمستخدم: @{message.from_user.username or message.from_user.first_name} (ID: {message.from_user.id})\nالمنتج: {prod_name}"
        keyboard = InlineKeyboardMarkup()
        keyboard.add(
            InlineKeyboardButton("قبول ومتابعة التسليم", callback_data=f"approve_{order_id}"),
            InlineKeyboardButton("رفض الطلب", callback_data=f"reject_{order_id}")
        )
        await bot.send_photo(ADMIN_ID, message.photo[-1].file_id, caption=caption, reply_markup=keyboard)

# موافقة أو رفض الطلب من الأدمن
@dp.callback_query_handler(lambda c: c.data.startswith('approve_') or c.data.startswith('reject_'))
async def process_admin_action(callback_query: types.CallbackQuery):
    if not is_admin(callback_query.from_user.id):
        return
    action, order_id_str = callback_query.data.split('_')
    order_id = int(order_id_str)

    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()

    if action == 'approve':
        cursor.execute("UPDATE orders SET status = 'approved' WHERE order_id = ?", (order_id,))
        conn.commit()
        await callback_query.message.reply(f"تم قبول الطلب #{order_id}. أرسل الكود الحقيقي للعميل بهذا الأمر:\n`/deliver {order_id} <code`", parse_mode="Markdown")
        await callback_query.answer("تم القبول.")
    else:
        cursor.execute("SELECT user_id FROM orders WHERE order_id = ?", (order_id,))
        res = cursor.fetchone()
        cursor.execute("UPDATE orders SET status = 'rejected' WHERE order_id = ?", (order_id,))
        conn.commit()
        if res:
            try:
                await bot.send_message(res[0], f"عذراً، تم رفض طلبك رقم #{order_id}. يرجى التواصل مع الدعم.")
            except Exception:
                pass
        await callback_query.message.reply(f"تم رفض الطلب #{order_id}.")
        await callback_query.answer("تم الرفض.")
    conn.close()

# تسليم الكود
@dp.message_handler(commands=['deliver'])
async def deliver_code(message: types.Message):
    if not is_admin(message.from_user.id):
        return
    parts = message.text.split(maxsplit=2)
    if len(parts) < 3:
        await message.reply("الاستخدام: /deliver <order_id> <real_code>")
        return
    order_id_str, real_code = parts[1], parts[2]
    try:
        order_id = int(order_id_str)
    except ValueError:
        await message.reply("رقم الطلب غير صحيح.")
        return

    conn = sqlite3.connect('store.db')
    cursor = conn.cursor()
    cursor.execute("SELECT user_id, status FROM orders WHERE order_id = ?", (order_id,))
    res = cursor.fetchone()

    if not res or res[1] != 'approved':
        await message.reply("الطلب غير موجود أو لم تتم الموافقة عليه بعد.")
        conn.close()
        return

    user_id = res[0]
    cursor.execute("UPDATE orders SET status = 'delivered' WHERE order_id = ?", (order_id,))
    conn.commit()
    conn.close()

    try:
        await bot.send_message(user_id, f"تمت الموافقة على طلبك. كودك:\n`{real_code}`", parse_mode="Markdown")
        await message.reply(f"تم إرسال كود الطلب #{order_id}.")
    except Exception as e:
        await message.reply(f"فشل إرسال الكود للعميل: {e}")

if __name__ == '__main__':
    executor.start_polling(dp, skip_updates=True)
